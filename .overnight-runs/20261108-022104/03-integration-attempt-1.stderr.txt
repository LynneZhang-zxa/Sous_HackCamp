export type RecipeQuery = { q?: string; cuisine?: string; maxMinutes?: string; tag?: string; sort?: string };
export function searchRecipes(recipes: readonly Recipe[], query: RecipeQuery): Recipe[] {
 const terms = (query.q ?? "").trim().toLowerCase().split(/\s+/).filter(Boolean);
 const max = Number(query.maxMinutes);
 return recipes.filter(r => {
  const text = [r.title, r.cuisine, ...r.tags, ...r.ingredients.map(i => i.name)].join(" ").toLowerCase();
  return terms.every(t => text.includes(t)) && (!query.cuisine || r.cuisine.toLowerCase() === query.cuisine.toLowerCase() || r.cuisine.toLowerCase().split(/[- ]/).includes(query.cuisine.toLowerCase())) && (!query.tag || r.tags.includes(query.tag)) && (!Number.isFinite(max) || max <= 0 || r.totalMinutes <= max);
 }).sort((a,b) => {
  const time = query.sort === "shortest" ? a.totalMinutes-b.totalMinutes : query.sort === "longest" ? b.totalMinutes-a.totalMinutes : 0;
  return time || a.title.localeCompare(b.title, "en") || a.id.localeCompare(b.id, "en");
 });
}

export async function POST(req: Request) {
  try {
    let body: unknown;
    try { body = await req.json(); } catch { return NextResponse.json({ error: "Invalid JSON" }, { status: 400 }); }
    if (!body || typeof body !== "object" || Array.isArray(body) || ("recipeId" in body && typeof body.recipeId !== "string")) {
      return NextResponse.json({ error: "recipeId must be a string" }, { status: 400 });
    }
    const { recipeId } = body as { recipeId?: string };
    const recipe = getRecipe(recipeId ?? CREAMY_GARLIC_CHICKEN.id);
    if (!recipe) return NextResponse.json({ error: "Unknown recipe" }, { status: 404 });

    const agentId = requireEnv("ALEBEX_AGENT_ID");
    const session = await createSession(recipe.id);
    const { base, source } = await resolvePublicBase(req);
    const warnings: string[] = [];
    if (!base) warnings.push("No public https base for tools (start ngrok or set PUBLIC_BASE). The agent can talk but cannot change steps.");

    let token: string;
    try {
      token = await mintWsToken();
    } catch (e) {
      if (process.env.NODE_ENV === "production" && process.env.ALLOW_RAW_KEY !== "1") throw e;
      console.warn("[sous] token mint failed, using raw key for the socket:", (e as Error).message);
      token = requireEnv("ALEBEX_API_KEY");
      // Dev-only fallback; logged above, not shown to the cook.
    }

    console.log(`[sous] call start session=${session.id} tools=${base ?? "none"} (${source})`);
    return NextResponse.json({
      sessionId: session.id,
      secret: session.secret,
      wsUrl: WS_URL,
      subprotocol: `alebex.token.${token}`,
      startCall: buildStartCall(agentId, recipe, session, base),
      state: session,
      recipe,
      warnings,
    });
  } catch (e) {
    const message = (e as Error).message;
    console.error("[sous] call start failed:", message);
    return NextResponse.json({ error: message }, { status: 500 });
  }
}

 const [ready,setReady] = useState(false);
 const [error,setError] = useState("");
 useEffect(() => {
  const refresh = () => { try { setIds(parseSaved(localStorage.getItem(SAVED_KEY))); } catch { setError("Browser storage is unavailable. Saved recipes cannot persist."); } setReady(true); };
  refresh(); window.addEventListener("storage",refresh); window.addEventListener("sous-saved",refresh);
  return () => { window.removeEventListener("storage",refresh); window.removeEventListener("sous-saved",refresh); };
 },[]);
 const toggle = (id: string) => { try { const next = toggleSaved(parseSaved(localStorage.getItem(SAVED_KEY)),id); localStorage.setItem(SAVED_KEY,JSON.stringify(next)); setIds(next); setError(""); window.dispatchEvent(new Event("sous-saved")); } catch { setError("Could not save your change. Browser storage is unavailable or full."); } };
 return {ids,ready,error,toggle};
}

export type Ingredient = {
  name: string;
  /** Amount for the recipe's base servings. Null for "to taste" items. */
  amount: number | null;
  unit: string; // "", "cup", "tablespoon", "teaspoon", "clove", "pound", "ounce"
  note?: string;
};

export type Step = {
  id: string;
  title: string;
  /** One to three plain sentences, written to be read aloud. */
  instruction: string;
  /** Suggested time for this step, used when the cook asks for "a timer" without a length. */
  durationSeconds?: number;
  tip?: string;
  safety?: string;
  /** Ingredient names used in this step, matched against Ingredient.name. */
  ingredients?: string[];
};

  id: string;

  totalMinutes: number;
  /** Legacy per-serving field. Null means unavailable; see nutrition for provenance. */
  calories: number | null;
  nutrition: { status: "unavailable"; reason: string } | {
    status: "estimated";
    basis: "per-serving";
    calories: number;
    source: string;
  };
  emoji: string;
  ingredients: Ingredient[];
  steps: Step[];
};

function parseUtterance(f: Frame): { role: "cook" | "sous"; text: string; final: boolean } | null {
  const text = [f.text, f.transcript, f.content, f.message, f.line].find((v) => typeof v === "string" && v.trim()) as string | undefined;
  if (!text) return null;
  const who = String(f.role ?? f.speaker ?? f.from ?? f.source ?? f.participant ?? "").toLowerCase();
  const role: "cook" | "sous" = /user|caller|human|you|customer/.test(who) ? "cook" : /agent|assistant|ai|bot|model/.test(who) ? "sous" : f.type === "transcript" ? "cook" : "sous";
  const finalRaw = f.final ?? f.is_final ?? f.isFinal ?? f.completed;
  return { role, text: text.trim(), final: finalRaw === undefined ? true : Boolean(finalRaw) };

+// Exercise actual pages and call-start without workers, network, or credentials.
+const fs = require('node:fs');
+const path = require('node:path');
+const ts = require('typescript');
+const assert = require('node:assert/strict');
+const React = require('react');
+const { renderToStaticMarkup } = require('react-dom/server');
+const root = path.resolve(__dirname, '..');
+const cache = new Map();
+let sessionIds = [];
+let tokenRequests = 0;
+function load(file) {
+  const full = path.resolve(root, file);
+  if (cache.has(full)) return cache.get(full).exports;
+  const module = { exports: {} }; cache.set(full, module);
+  const code = ts.transpileModule(fs.readFileSync(full, 'utf8'), {
+    compilerOptions: { module: ts.ModuleKind.CommonJS, jsx: ts.JsxEmit.ReactJSX, target: ts.ScriptTarget.ES2022 },
+    fileName: full,
+  }).outputText;
+  function localRequire(name) {
+    if (name === 'next/navigation') return { notFound() { throw new Error('404'); } };
+    if (name === '@/components/cook/CookScreen') return { CookScreen: () => null };
+    if (name === '@/lib/store') return { createSession: async id => {
+      sessionIds.push(id); return load('lib/session.ts').newSession(id, 'offline', 'offline');
+    } };
+    if (name === '@/lib/alebex') return {
+      requireEnv: () => 'offline-agent', WS_URL: 'wss://example.test',
+      resolvePublicBase: async () => ({ base: 'https://example.test', source: 'offline' }),
+      mintWsToken: async () => { tokenRequests++; return 'offline-token'; },
+    };
+    if (!name.startsWith('.') && !name.startsWith('@/')) return require(name);
+    const base = name.startsWith('@/') ? path.resolve(root, name.slice(2)) : path.resolve(path.dirname(full), name);
+    const target = ['.ts', '.tsx'].map(ext => base + ext).find(fs.existsSync);
+    if (!target) throw new Error('Unresolved local module: ' + name);
+    return load(target);
+  }
+  new Function('require', 'module', 'exports', code)(localRequire, module, module.exports);
+  return module.exports;
+}
+async function main() {
+  const { listRecipes, CREAMY_GARLIC_CHICKEN } = load('lib/recipe.ts');
+  const { searchRecipes } = load('lib/recipe-search.ts');
+  const { parseSaved, toggleSaved } = load('lib/saved-recipes.ts');
+  const detail = load('app/recipes/[id]/page.tsx');
+  const cook = load('app/cook/page.tsx').default;
+  const listing = load('app/recipes/page.tsx').default;
+  const { POST } = load('app/api/call/start/route.ts');
+  const recipes = listRecipes(); assert.equal(recipes.length, 50);
+  assert.deepEqual(detail.generateStaticParams(), recipes.map(({ id }) => ({ id })));
+  const html = renderToStaticMarkup(await listing({ searchParams: Promise.resolve({}) }));
+  let saved = [];
+  for (const recipe of recipes) {
+    assert.ok(html.includes(`href="/recipes/${recipe.id}"`));
+    const page = renderToStaticMarkup(await detail.default({ params: Promise.resolve({ id: recipe.id }) }));
+    assert.ok(page.includes(`href="/cook?recipeId=${recipe.id}"`));
+    assert.equal((await cook({ searchParams: Promise.resolve({ recipeId: recipe.id }) })).props.selectedRecipe.id, recipe.id);
+    assert.ok(searchRecipes(recipes, { q: recipe.title.toUpperCase(), cuisine: recipe.cuisine, tag: recipe.tags[0], maxMinutes: String(recipe.totalMinutes) }).includes(recipe));
+    saved = toggleSaved(saved, recipe.id);
+    const response = await POST(new Request('https://example.test/api/call/start', { method: 'POST', body: JSON.stringify({ recipeId: recipe.id }) }));
+    assert.equal(response.status, 200);
+    const body = await response.json();
+    assert.equal(body.recipe.id, recipe.id); assert.equal(body.state.recipeId, recipe.id);
+    assert.ok(body.startCall.context.includes(recipe.title));
+  }
+  assert.deepEqual(parseSaved(JSON.stringify(saved)), recipes.map(r => r.id));
+  for (const recipe of recipes) assert.deepEqual(toggleSaved(saved, recipe.id), saved.filter(id => id !== recipe.id));
+  assert.deepEqual(sessionIds, recipes.map(r => r.id)); assert.equal(tokenRequests, 50);
+  assert.equal((await cook({ searchParams: Promise.resolve({}) })).props.selectedRecipe.id, CREAMY_GARLIC_CHICKEN.id);
+  for (const id of ['', 'missing', '__proto__', 'toString']) {
+    await assert.rejects(detail.default({ params: Promise.resolve({ id }) }), /404/);
+    await assert.rejects(cook({ searchParams: Promise.resolve({ recipeId: id }) }), /404/);
+    assert.equal((await POST(new Request('https://example.test/api/call/start', { method: 'POST', body: JSON.stringify({ recipeId: id }) }))).status, 404);
+  }
+  assert.equal(tokenRequests, 50);
+  console.log('PASS: all 50 listing/detail/cook selections, combined search, full favorites collection, mocked call-start IDs, default cook and invalid-ID 404 handling. No provider calls.');
+}
+main().catch(error => { console.error(error); process.exitCode = 1; });
